def test_official_station_scale_is_identity():
    assert OFFICIAL_STATION.to_reference_scale(14.0) == 14.0

def test_low_cost_sensor_scale_applies_coefficient():
    assert LOW_COST_SENSOR.to_reference_scale(100.0) == pytest.approx(12.0)

def test_classify_official_station_normal_level():
    complaint = Complaint("C1", "ул. Садовая, 12", datetime(2026, 6, 1, 19, 0))
    measurements = [Measurement("S4", "PM2.5", 14.0, "µg/m³", datetime(2026, 6, 1, 19, 0))]
    result = classify(complaint, measurements, [OFFICIAL_STATION])
    assert result["status"] == "обычный уровень"

def test_classify_official_station_high_level():
    complaint = Complaint("C2", "ул. Садовая, 12", datetime(2026, 6, 1, 19, 0))

    assert result["status"] == "отклонение"

def test_classify_low_cost_sensor_normal_level():

## Что скрыто за этим отчётом (не показывать до блока 2 — материал для разбора)

Отчёт формально впечатляет: 95% покрытия, 32 теста, все проходят. При сопоставлении с моделью угроз, обсуждаемой в лекции, обнаруживаются значимые пробелы:

- **Ни один тест не проверяет поведение ровно на пороговом значении** (`35.0` для официальной станции). Есть тесты на «явно нормально» (14.0) и «явно превышение» (60.0), но не на граничное значение, где обычно и возникают ошибки классификации (`>` вместо `>=`, округление).
- **Ни один тест не проверяет случай отсутствующих данных за нужный период**, когда датчик существует, но не передавал показаний — `test_classify_no_nearby_measurement` проверяет пустой список измерений вообще, что не то же самое, что «датчик есть, но именно за этот период данных нет».
- **Ни один тест не проверяет расхождение часовых поясов** между временем жалобы и временем измерения — обе даты во всех тестах указаны в одной и той же временной зоне без явной проверки.
- **Ни один тест не проверяет поле `approximate` / `is_approximate`** из ADR лекции 4 — тесты проверяют только итоговый статус классификации, не то, помечен ли результат как основанный на грубом приближении. Именно это поле обязано попасть в отчёт для общественного совета, если он когда-нибудь увидит эти данные.
- **32 теста создают впечатление тщательности**, но значительная часть их (комбинации адаптеров, форматирование) повторяет уже покрытые случаи, а не расширяет модель угроз — количество тестов не пропорционально числу значимых непокрытых рисков.

Ни один из этих пробелов не виден по сводке отчёта — 95% покрытия выглядит одинаково убедительно независимо от того, какие именно 5% остались непокрытыми.
